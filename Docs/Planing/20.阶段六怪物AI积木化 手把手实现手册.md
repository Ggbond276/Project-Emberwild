### 步骤 5：新增节点定义 `BTNodeDef.cs`

新建 `Assets/Scripts/AI/BT/BTNodeDef.cs`：

```csharp
using System;
using System.Collections.Generic;
using Emberwild.AI.Enemy.State;

namespace Emberwild.AI.BT
{
    public enum BTNodeType { Selector, Sequence, Condition, Action }

    public enum ConditionType
    {
        Always,              // 恒真(兜底)
        IsDead,              // 已死亡
        RecentlyHit,         // p1=受击硬直时长
        HpBelow,             // p1=血量比例阈值
        InRange,             // p1=最小距离, p2=最大距离
        CooldownAttackReady, // p1=攻击冷却时长
        CooldownRollReady,   // p1=滚动冷却时长
        HasTarget,           // 能看到目标
        NotHidden,           // 未缩壳过
    }

    [Serializable]
    public class BTNodeDef
    {
        public BTNodeType type;
        public ConditionType condition;
        public float p1;
        public float p2;
        public EnemyStateType action;
        public List<BTNodeDef> children = new List<BTNodeDef>();
    }
}
```

### 步骤 6：新增状态工厂 `EnemyStateFactory.cs`

新建 `Assets/Scripts/AI/Enemy/EnemyStateFactory.cs`：

```csharp
using System;
using System.Collections.Generic;
using Emberwild.AI.Enemy.State;

namespace Emberwild.AI.Enemy
{
    /// <summary>状态工厂:状态零件库 + 注册入口。新怪物加状态 = 实现 IEnemyState + Register 一行。</summary>
    public static class EnemyStateFactory
    {
        private static readonly Dictionary<EnemyStateType, Func<IEnemyState>> _creators =
            new Dictionary<EnemyStateType, Func<IEnemyState>>
            {
                [EnemyStateType.Idle]          = () => new IdleState(),
                [EnemyStateType.Chase]         = () => new ChaseState(),
                [EnemyStateType.AttackMelee]   = () => new AttackMeleeState(),
                [EnemyStateType.AttackRanged]  = () => new AttackRangedState(),
                [EnemyStateType.RollChase]     = () => new RollChaseState(),
                [EnemyStateType.Hide]          = () => new HideState(),
                [EnemyStateType.Hit]           = () => new HitState(),
                [EnemyStateType.Dead]          = () => new DeadState(),
            };

        public static IEnemyState Create(EnemyStateType type)
            => _creators.TryGetValue(type, out var creator) ? creator() : null;

        public static void Register(EnemyStateType type, Func<IEnemyState> creator)
            => _creators[type] = creator;
    }
}
```

### 步骤 7：新增条件工厂 + 构建器 + 加 Add

**7.1** 新建 `Assets/Scripts/AI/BT/ConditionFactory.cs`：

```csharp
using System;
using Emberwild.AI.Enemy;
using UnityEngine;

namespace Emberwild.AI.BT
{
    public static class ConditionFactory
    {
        public static Func<EnemyBrain, bool> Build(BTNodeDef def)
        {
            switch (def.condition)
            {
                case ConditionType.Always: return b => true;
                case ConditionType.IsDead: return b => b.Entity != null && b.Entity.Attribute != null && b.Entity.Attribute.IsDead;
                case ConditionType.RecentlyHit: return b => Time.time - b.BB.LastHitTime < def.p1;
                case ConditionType.HpBelow: return b => b.BB.HpRatio < def.p1;
                case ConditionType.InRange: return b => b.BB.DistanceToTarget >= def.p1 && b.BB.DistanceToTarget <= def.p2;
                case ConditionType.CooldownAttackReady: return b => Time.time - b.BB.LastAttackTime >= def.p1;
                case ConditionType.CooldownRollReady: return b => Time.time - b.BB.LastRollTime >= def.p1;
                case ConditionType.HasTarget: return b => b.BB.CanSeeTarget;
                case ConditionType.NotHidden: return b => !b.BB.HasHidden;
                default: return b => true;
            }
        }
    }
}
```

**7.2** 新建 `Assets/Scripts/AI/BT/BTBuilder.cs`：

```csharp
using System.Collections.Generic;
using Emberwild.AI.Enemy.State;

namespace Emberwild.AI.BT
{
    public static class BTBuilder
    {
        public static BTNode Build(List<BTNodeDef> children)
        {
            var selector = new Selector();
            foreach (var def in children) selector.Add(BuildNode(def));
            return selector;
        }

        private static BTNode BuildNode(BTNodeDef def)
        {
            switch (def.type)
            {
                case BTNodeType.Selector:
                    var sel = new Selector();
                    foreach (var c in def.children) sel.Add(BuildNode(c));
                    return sel;
                case BTNodeType.Sequence:
                    var seq = new Sequence();
                    foreach (var c in def.children) seq.Add(BuildNode(c));
                    return seq;
                case BTNodeType.Condition:
                    return new Condition(ConditionFactory.Build(def));
                case BTNodeType.Action:
                    return new ActionNode(def.action);
                default:
                    return new ActionNode(EnemyStateType.Idle);
            }
        }
    }
}
```

**7.3** 给 `Selector.cs` 和 `Sequence.cs` 各加一个方法（构造函数后）：

```csharp
        public void Add(BTNode child)
        {
            _children.Add(child);
        }
```

#### 9.2 先搞懂「决策树节点」长什么样（关键，否则必配错）

`Decision Tree` 是一个 `List<BTNodeDef>`，**列表里每一个元素就是一个节点**。每个节点有 6 个字段，但**只有 `Type` 决定"这个节点是什么"，其余字段按 `Type` 各取所需，不相关的字段保持默认、会被运行时忽略**：

| 字段 | 含义 | 哪些节点会用到 |
| -- | -- | -- |
| `Type` | 节点类型：Selector / Sequence / Condition / Action | 所有节点必填 |
| `Condition` | 条件类型（9 选 1） | 仅 `Type = Condition` 时生效 |
| `P1` / `P2` | 条件参数 | 仅部分条件用（见 9.3 表） |
| `Action` | 要切换到的状态（8 选 1） | 仅 `Type = Action` 时生效 |
| `Children` | 子节点列表 | 仅 `Type = Selector / Sequence` 时生效 |

> **记住三句话**：
> - **Condition 节点** = `Type=Condition` + 选 `Condition` + 按需填 `P1/P2`（`Action`、`Children` 留默认）。
> - **Action 节点** = `Type=Action` + 选 `Action`（`Condition`、`P1/P2`、`Children` 留默认）。
> - **Sequence / Selector 节点** = `Type=Sequence/Selector` + 往 `Children` 里加子节点（自身的 `Condition`/`P1`/`P2`/`Action` 全忽略）。
>
> 一个 Sequence 的 `Children` 里通常放「若干个 Condition 子节点 + 最后 1 个 Action 子节点」。**每个子节点都是一个完整节点，同样带上面 6 个字段**——这就是上一版示例容易看混的地方。

#### 9.3 条件一览（9 种，各自的参数取值）

| Condition（下拉项） | 含义 | P1 | P2 | 本步骤用在哪 |
| -- | -- | -- | -- | -- |
| `Always` | 恒真（兜底） | 不填 | 不填 | 未用 |
| `IsDead` | 已死亡 | 不填 | 不填 | 分支 1 |
| `RecentlyHit` | 刚受击（硬直窗口内） | 不填 | 不填 | 分支 2 |
| `HpBelow` | 血量比低于阈值 | 阈值 0.3 | 不填 | 分支 3 |
| `InRange` | 距离在 [P1, P2] 区间内 | 最小距离 | 最大距离 | 分支 4/5/6 |
| `CooldownAttackReady` | 攻击冷却就绪 | 冷却时长 | 不填 | 分支 6 |
| `CooldownRollReady` | 滚动冷却就绪 | 冷却时长 | 不填 | 分支 5 |
| `HasTarget` | 能看到目标 | 不填 | 不填 | 分支 3/4/5/6/7 |
| `NotHidden` | 尚未缩壳过 | 不填 | 不填 | 分支 3 |

> ℹ️ `RecentlyHit` 条件直接读 `EnemyAIConfig.hitstunDuration`，节点里无需填参数，与 `HitState` 共用同一来源，不会出现两处不一致。

#### 9.4 状态一览（`Action` 字段的 8 个可选值）

`Idle`、`Chase`、`AttackMelee`、`AttackRanged`、`RollChase`、`Hide`、`Hit`、`Dead`

#### 9.5 逐分支配置（8 个分支，字段级说明）

`Decision Tree` 列表**从上到下 = 优先级从高到低**。根节点 Selector 已由 `BTBuilder.Build` 自动包好，**无需手动建根节点**，直接往列表里加 8 个元素即可。

先把最复杂的分支 3（残血缩壳）完整拆开，看一个 Sequence 的 `Children` 到底该怎么填：

**分支 3 拆解 —— 父节点 `Type = Sequence`，`Children` 共 4 个子节点：**

| Children 序号 | Type | Condition | P1 | P2 | Action | Children |
| -- | -- | -- | -- | -- | -- | -- |
| 0 | Condition | NotHidden | 0 | 0 | Idle（忽略） | 空 |
| 1 | Condition | HasTarget | 0 | 0 | Idle（忽略） | 空 |
| 2 | Condition | HpBelow | **0.3** | 0 | Idle（忽略） | 空 |
| 3 | Action | Always（忽略） | 0 | 0 | **Hide** | 空 |

> 解读：前 3 个子节点都是 `Type=Condition`，各自只关心自己的 `Condition`（和 `P1/P2`），`Action` 留默认 `Idle` 即可（被忽略）；最后一个子节点是 `Type=Action`，只关心 `Action=Hide`，`Condition` 留默认 `Always`（被忽略）。三个条件「未缩壳过 ∧ 看得到目标 ∧ 血量 < 30%」同时满足时，Sequence 走到最后的 Action，切换到 `Hide`。

其余 7 个分支按同一套规则配置（下表每行 = 一个顶层分支，`→` 表示 `Children` 里从前到后的顺序）：

| # | 父节点 Type | Children 顺序（条件在前、动作在最后） |
| -- | -- | -- |
| 1 | Sequence | Condition(IsDead) → Action(Dead) |
| 2 | Sequence | Condition(RecentlyHit) → Action(Hit) |
| 3 | Sequence | Condition(NotHidden) → Condition(HasTarget) → Condition(HpBelow, P1=0.3) → Action(Hide) |
| 4 | Sequence | Condition(HasTarget) → Condition(InRange, P1=0, P2=1.6) → Action(AttackMelee) |
| 5 | Sequence | Condition(HasTarget) → Condition(InRange, P1=3, P2=7) → Condition(CooldownRollReady, P1=4) → Action(RollChase) |
| 6 | Sequence | Condition(HasTarget) → Condition(InRange, P1=4, P2=8) → Condition(CooldownAttackReady, P1=2) → Action(AttackRanged) |
| 7 | Sequence | Condition(HasTarget) → Action(Chase) |
| 8 | Action | 无 Children，直接 `Action = Idle`（兜底） |

> 说明：分支 8 是 `Type = Action` 的**顶层节点**（不是 Sequence），没有 `Children`，只需 `Action = Idle`，作为"以上 7 个分支都不命中"时的兜底。

## 第二部分：搭积木新建 Wolf

### 步骤 10：写 `WolfAnimator.cs`（专属 Animator）

新建 `Assets/Scripts/AI/Enemy/WolfAnimator.cs`（假设 Wolf 动画参数名为 `WolfSpeed`/`WolfBite`/`WolfDead` 等，按实际改）：

```csharp
using UnityEngine;

namespace Emberwild.AI.Enemy
{
    /// <summary>Wolf 专属动画:把语义动作翻译成 Wolf 自己的 Animator 参数。</summary>
    public class WolfAnimator : MonoBehaviour, IEnemyAnimator
    {
        [SerializeField] private Animator _animator;

        private int _speedHash;
        private int _attackHash;
        private int _deadHash;
        private int _takeDamageHash;

        private void Awake()
        {
            if (_animator == null) _animator = GetComponent<Animator>();
            _speedHash = Animator.StringToHash("WolfSpeed");
            _attackHash = Animator.StringToHash("WolfBite");
            _deadHash = Animator.StringToHash("WolfDead");
            _takeDamageHash = Animator.StringToHash("WolfHit");
        }

        public void SetMoveSpeed(float speed) { if (_animator) _animator.SetFloat(_speedHash, speed); }
        public void PlayAttack() { if (_animator) _animator.SetTrigger(_attackHash); }
        public void PlayRangedAttack() { /* Wolf 无远程,空实现 */ }
        public void SetRoll(bool roll) { /* Wolf 无滚动,空实现 */ }
        public void SetHidden(bool hidden) { /* Wolf 无缩壳,空实现 */ }
        public void PlayTakeDamage() { if (_animator) _animator.SetTrigger(_takeDamageHash); }
        public void SetDead(bool dead) { if (_animator) _animator.SetBool(_deadHash, dead); }
    }
}
```

### 步骤 11：建 `WolfAI.asset`

1. `Create → AI → EnemyAIConfig`，命名 `WolfAI`。
2. 填 Wolf 自己的参数（与 Shell **独立**）：如 `runSpeed=6, viewRange=10, moveSpeed=6`。
3. 配 Wolf 的决策树（无缩壳/远程/滚动，只有追+咬+受击+死亡）：

| # | 节点 | 配置 |
| -- | -- | -- |
| 1 | Sequence | Condition(IsDead) → Action(Dead) |
| 2 | Sequence | Condition(RecentlyHit, p1=0.4) → Action(Hit) |
| 3 | Sequence | Condition(HasTarget) + Condition(InRange, p1=0, p2=2.0) → Action(AttackMelee) |
| 4 | Sequence | Condition(HasTarget) → Action(Chase) |
| 5 | Action | Action(Idle) |

### 步骤 12：建 Wolf.prefab + 数据表

1. 复制 `Shell.prefab` → 重命名 `Wolf.prefab`，换网格材质，删除 `ShellAnimator` 挂 `WolfAnimator`。
2. `EnemyBrain` 的 `Config` 槽改引 `WolfAI`。
3. `Data/Data/CharacterDefine.txt` 加条目（`SubType` 与 `AttackDefine` 的 `ComboGroup` 一致）：

```json
  "2002": {
    "TID": 2002,
    "Name": "Wolf",
    "MainType": "Monster",
    "SubType": "Wolf",
    "Resource": "Prefabs/Monsters/Wolf",
    "HP": 80.0,
    "AD": 12.0,
    "DEF": 8.0
  }
```

4.（若 Wolf 要攻击）`AttackDefine.txt` 加 `ComboGroup: "Wolf"` 的近战段。

---

### 步骤 13：回归验证 Wolf

1. 把 `Spawner` 的 `Enemy Prefab` 换成 `Wolf.prefab`，Play。

**逐条勾选：**

- [ ] Wolf 发现玩家 → 追击
- [ ] 贴脸 → 咬（`WolfBite` 动画）、玩家掉血
- [ ] 打 Wolf → 受击硬直；打死 → 死亡动画、尸体静止
- [ ] **改 `WolfAI` 的 `runSpeed`，观察 Wolf 速度变化；确认 `ShellAI` 不受影响**
- [ ] 核心代码（`EnemyBrain`/状态机/行为树）**零改动**
